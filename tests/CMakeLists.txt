cmake_minimum_required(VERSION 3.16)
project(tabnamp_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JSON_ROOT "$ENV{HOME}/esp/esp-idf/components/json/cJSON")
add_executable(core_tests core_tests.cpp ../core/mp3_seek.cpp ../core/mp4_stream.cpp ../core/wav.cpp ../core/eq.cpp ../core/playlist.cpp ../core/storage.cpp ../core/metadata.cpp ../core/spectrum.cpp ../core/hid.cpp ../core/hls.cpp ../core/library.cpp "${JSON_ROOT}/cJSON.c")
set_source_files_properties("${JSON_ROOT}/cJSON.c" PROPERTIES COMPILE_OPTIONS "-Wno-deprecated-declarations")
target_include_directories(core_tests PRIVATE ../core "${JSON_ROOT}")
target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(core_tests PRIVATE -fsanitize=address,undefined)
enable_testing()
add_test(NAME core COMMAND core_tests)
add_executable(mp4_view mp4_view.cpp ../core/mp4_stream.cpp)
target_include_directories(mp4_view PRIVATE ../core)
target_compile_options(mp4_view PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(mp4_view PRIVATE -fsanitize=address,undefined)
add_executable(engine_tests engine_tests.cpp ../core/engine.cpp ../core/eq.cpp ../core/storage.cpp ../core/playlist.cpp ../core/metadata.cpp ../core/library.cpp ../core/wav.cpp ../core/spectrum.cpp "${JSON_ROOT}/cJSON.c")
target_include_directories(engine_tests PRIVATE ../core "${JSON_ROOT}")
target_compile_options(engine_tests PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers -fsanitize=address,undefined)
target_link_options(engine_tests PRIVATE -fsanitize=address,undefined)
add_test(NAME engine_cancellation COMMAND engine_tests)
add_executable(mp3_index mp3_index.cpp ../core/mp3_seek.cpp)
target_include_directories(mp3_index PRIVATE ../core)
target_compile_options(mp3_index PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(mp3_index PRIVATE -fsanitize=address,undefined)
