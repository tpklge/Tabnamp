cmake_minimum_required(VERSION 3.16)
project(tabnamp_legacy_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.tools/ffmpeg-host-build/install")
add_executable(legacy_tests main.cpp ../../platform/host/decoder.cpp ../../core/playlist.cpp)
target_include_directories(legacy_tests PRIVATE ../../core "${ROOT}/include")
target_link_libraries(legacy_tests PRIVATE "${ROOT}/lib/libavformat.a" "${ROOT}/lib/libavcodec.a" "${ROOT}/lib/libswresample.a" "${ROOT}/lib/libavutil.a" m)
if(APPLE)
    target_link_libraries(legacy_tests PRIVATE "-framework CoreFoundation" "-framework CoreVideo" "-framework CoreMedia" "-framework Security" iconv)
endif()
target_compile_options(legacy_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(legacy_tests PRIVATE -fsanitize=address,undefined)
