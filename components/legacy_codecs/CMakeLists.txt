set(LEGACY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.tools/ffmpeg-p4-build/install")
if(NOT EXISTS "${LEGACY_ROOT}/lib/libavcodec.a")
    message(FATAL_ERROR "Run tools/build_legacy_decoders.sh inside the ESP-IDF environment first (tools/build.sh does this automatically).")
endif()
idf_component_register(SRCS "../../platform/host/decoder.cpp" INCLUDE_DIRS "../../core" "${LEGACY_ROOT}/include")
target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
target_compile_definitions(${COMPONENT_LIB} PRIVATE TABNAMP_DECODER_FACTORY=make_legacy_decoder TABNAMP_LEGACY_ONLY=1)
foreach(lib avformat avcodec swresample avutil)
    add_prebuilt_library(tabnamp_${lib} "${LEGACY_ROOT}/lib/lib${lib}.a")
endforeach()
target_link_libraries(${COMPONENT_LIB} PRIVATE tabnamp_avformat tabnamp_avcodec tabnamp_swresample tabnamp_avutil m)
