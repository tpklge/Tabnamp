cmake_minimum_required(VERSION 3.16)
project(tabnamp_emulator LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
option(TABNAMP_SANITIZE "ASan/UBSan" ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libswresample libavutil)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory(../managed_components/lvgl__lvgl lvgl)
set(JSON_ROOT "$ENV{IDF_PATH}/components/json/cJSON")
if(NOT EXISTS "${JSON_ROOT}/cJSON.c")
    set(JSON_ROOT "$ENV{HOME}/esp/esp-idf/components/json/cJSON")
endif()
set_source_files_properties("${JSON_ROOT}/cJSON.c" PROPERTIES COMPILE_OPTIONS "-Wno-deprecated-declarations")
add_library(tabnamp_core ../core/mp3_seek.cpp ../core/mp4_stream.cpp ../core/wav.cpp ../core/eq.cpp ../core/media.cpp ../core/metadata.cpp ../core/playlist.cpp ../core/storage.cpp ../core/spectrum.cpp ../core/hid.cpp ../core/hls.cpp ../core/library.cpp ../core/engine.cpp ../platform/host/decoder.cpp "${JSON_ROOT}/cJSON.c")
target_include_directories(tabnamp_core PUBLIC ../core "${JSON_ROOT}")
target_link_libraries(tabnamp_core PUBLIC PkgConfig::AV)
target_compile_options(tabnamp_core PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
add_executable(tabnamp_emulator main.cpp ../ui/ui.cpp)
target_include_directories(tabnamp_emulator PRIVATE ../ui)
target_link_libraries(tabnamp_emulator PRIVATE tabnamp_core lvgl PkgConfig::SDL PkgConfig::CURL)
if(TABNAMP_SANITIZE)
    foreach(t tabnamp_core tabnamp_emulator lvgl)
        target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(tabnamp_emulator PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME emulator_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../tools/emulator_runner.py $<TARGET_FILE:tabnamp_emulator> ${CMAKE_CURRENT_BINARY_DIR}/fixtures ${CMAKE_CURRENT_BINARY_DIR}/artifacts)
set_tests_properties(emulator_integration PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;ASAN_OPTIONS=detect_leaks=0")
